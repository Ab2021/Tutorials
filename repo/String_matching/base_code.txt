def normalize_city_name(city):
 # Remove dots, extra spaces, trim, uppercase
 city = city.replace('.', '').replace('-', ' ').strip()
 city = re.sub(r'\s+', ' ', city) # Collapse multiple spaces
 return city.upper()

def extract_state(row,base_state_col,second_state_col):
 unit_number = row[base_state_col]
 unit_name = row[second_state_col]
# Extract prefix: all letters until first numeric digit
 prefix_match = re.match(r'^([A-Za-z]+)', unit_number)
 prefix = prefix_match.group(1).upper() if prefix_match else ''
if len(prefix) == 2:
 return prefix
# Try to get two-letter state code after comma in unit name
 match = re.search(r',\s*([A-Z]{2})\b', unit_name)
 if match:
 return match.group(1)
# Try to get city from unit_name (first word before comma, space, or end)
 city_match = re.match(r'^([A-Za-z .-]+?)(?=,| |\Z)', unit_name)
 if city_match:
 city = city_match.group(1).strip()
 city_norm = normalize_city_name(city) # << Normalization here!
 if city_norm in us_city_st_dict_norm:
 return us_city_st_dict_norm[city_norm]
# If not found, return full prefix as extracted (not truncated)
 return prefix

def parse_date(s, dayfirst=True):
 """
 Robust date parser: handles string (date, datetime, ISO8601, etc),
 pd.Timestamp, pandas NaT, datetime.datetime, datetime.date.
 Returns Python date or None.
 """
 # Handle pd.NaT and pd.NA
 if s is None or pd.isna(s):
 return None
 # If input is already datetime.date or datetime.datetime, return date part
 if isinstance(s, datetime):
 return s.date()
 if isinstance(s, date):
 return s
 # Try to parse string
 try:
 s_str = str(s)
 if not s_str.strip():
 return None
 dt = dateparser.parse(s_str, dayfirst=dayfirst)
 if dt is not None:
 return dt.date()
 except Exception:
 pass
 return None

def extract_X_coverage(coverage_str):
 if pd.isna(coverage_str):
 return ""
 s = str(coverage_str).strip()
 match = re.match(r"^\d+\s*-\s*([^-]+)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if match:
 cov = match.group(1).strip()
 excess = match.group(2)
 if excess:
 cov += " Excess"
 return cov
 s = re.sub(r"^\d+\s*-\s*", "", s)
 excess_match = re.search(r"(.*?)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if excess_match:
 cov = excess_match.group(1).strip()
 excess = excess_match.group(2)
 if excess:
 cov += " Excess"
 return cov
 s = re.sub(r"-\s*X.*$", "", s, flags=re.IGNORECASE).strip()
 return s

 return df

def extract_X_coverage(coverage_str):
 """
 Extract standard coverage label. Handles distinctions for 'Excess' lines.
 Examples:
 '10 -Workers Compensation - X' → 'Workers Compensation'
 '30 -Auto Liability - X' → 'Auto Liability'
 """
 if pd.isna(coverage_str):
 return ""
 s = str(coverage_str).strip()
# Pattern: num - Coverage - X (or X) [optional 'Excess']
 match = re.match(r"^\d+\s*-\s*([^-]+)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if match:
 cov = match.group(1).strip()
 excess = match.group(2)
 if excess:
 cov += " Excess"
 return cov
# Handle other variant, e.g. missing numeric prefix, or 'X' capitalization
 s = re.sub(r"^\d+\s*-\s*", "", s)
 excess_match = re.search(r"(.*?)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if excess_match:
 cov = excess_match.group(1).strip()
 excess = excess_match.group(2)
 if excess:
 cov += " Excess"
 return cov
# Fallback: just before any dash or trailing 'X' text
 s = re.sub(r"-\s*X.*$", "", s, flags=re.IGNORECASE).strip()
 return s
def assign_confidence(df):
 conditions = [
 (df['name_string_sim'] < 0.5),
 ((df['name_string_sim'] >= 0.5) & (df['name_string_sim'] < 0.54) & (df['description_string_sim'] < 0.1)),
 ((df['name_string_sim'] >= 0.54) & (df['description_string_sim'] >= 0.19)),
 ((df['name_string_sim'] >= 0.65) & (df['description_string_sim'] >= 0.04) & (df['description_string_sim'] < 0.19)),
 ((df['name_string_sim'] >= 0.5) & (df['name_string_sim'] < 0.54)),
 ]
 choices = ["Low (Name is low match)", "Medium", "High","High (Description is low match)","Low"]
 df['post_process_confidence'] = np.select(conditions, choices, default='Low')
 return df

def add_mean_group_desc_similarity(df):
 """
 For each X_clm_no_base_match_orig, compute for each row the mean similarity
 to other tpa_desc in the group (NaN if only 1 in group).
 Return a DataFrame with an added column: 'mean_desc_similarity_within_group'
 Does NOT deduplicate or drop rows!
 """
 def per_group(group):
 descs = group['tpa_desc'].fillna("").astype(str).tolist()
 n = len(descs)
 if n == 1:
 group = group.copy()
 group['mean_desc_similarity_within_group'] = 1.00
 return group
 # This matrix is (n x n)
 tfidf = TfidfVectorizer(stop_words='english', min_df=1)
 mat = tfidf.fit_transform(descs)
 sim_mat = cosine_similarity(mat)
# For each claim/row, compute mean similarity to all others (exclude self)
 means = []
 for i in range(n):
 others = [sim_mat[i, j] for j in range(n) if j != i]
 mean_sim = round(np.mean(others),2) if others else np.nan
 means.append(mean_sim)
 group = group.copy()
 group['mean_desc_similarity_within_group'] = means
 return group
return (
 df.groupby('X_clm_no_base_match_orig', group_keys=False)
 .apply(per_group)
 )

def extract_state(row,base_state_col,second_state_col):
 unit_number = row[base_state_col]
 unit_name = row[second_state_col]
# Extract prefix: all letters until first numeric digit
 prefix_match = re.match(r'^([A-Za-z]+)', unit_number)
 prefix = prefix_match.group(1).upper() if prefix_match else ''
if len(prefix) == 2:
 return prefix
# Try to get two-letter state code after comma in unit name
 match = re.search(r',\s*([A-Z]{2})\b', unit_name)
 if match:
 return match.group(1)
# Try to get city from unit_name (first word before comma, space, or end)
 city_match = re.match(r'^([A-Za-z .-]+?)(?=,| |\Z)', unit_name)
 if city_match:
 city = city_match.group(1).strip()
 city_norm = normalize_city_name(city) # << Normalization here!
 if city_norm in us_city_st_dict_norm:
 return us_city_st_dict_norm[city_norm]
# If not found, return full prefix as extracted (not truncated)
 return prefix

def parse_date(s, dayfirst=True):
 """
 Robust date parser: handles string (date, datetime, pd.Timestamp, pandas NaT, datetime.datetime, datetime.date).
 Returns pandas.Timestamp or pd.NaT.
 """
 import pandas as pd
 from datetime import date, datetime
 from dateutil import parser as dateparser
if s is None or pd.isna(s):
 return pd.NaT
 if isinstance(s, pd.Timestamp):
 return s
 if isinstance(s, datetime):
 return pd.Timestamp(s)
 if isinstance(s, date):
 return pd.Timestamp(s)
 try:
 s_str = str(s)
 if not s_str.strip():
 return pd.NaT
 dt = dateparser.parse(s_str, dayfirst=dayfirst)
 if dt is not None:
 return pd.Timestamp(dt)
 except Exception:
 pass
 return pd.NaT

 if not d1 or not d2:

def get_base_claim(claim_no):
 """
 Extracts the BASE claim:
 1. If it ends with letters + digits (e.g., …GB01), strip trailing digits (result: …GB)
 2. Else if it ends with a single letter (e.g., …49A), strip trailing letter (result: …49)
 3. Finally, if there is a dash ('-'), take the portion before it
 """
 claim_core = str(claim_no).strip()
 # Step 1: Remove trailing digits if immediately following uppercase letters (e.g. GB01 -> GB)
 m = re.match(r'^(.*?[A-Z]+)[0–9]+$', claim_core, re.IGNORECASE)
 if m:
 claim_core = m.group(1)
 # Step 2: Remove single trailing letter (e.g. …49A -> …49)
 elif re.match(r'.*[A-Z]$', claim_core, re.IGNORECASE):
 claim_core = re.sub(r'([A-Z])$', '', claim_core, flags=re.IGNORECASE)
 # Step 3: If dash present, take portion before dash
 if '-' in claim_core:
 claim_core = claim_core.split('-')[0].strip()
 return claim_core
def is_valid_pair(row_a, row_b, date_tolerance_days=0):
 # Ensure all filters (including line code!)
 state_matches = row_a['tpa_state'] == row_b['tpa_state']
 date_matches = abs(date_diff_days(
 parse_date(row_a['tpa_dt_loss']),
 parse_date(row_b['tpa_dt_loss'])

 line_matches = (
 normalize_text(row_a.get('tpa_line_cd', ''))
 == normalize_text(row_b.get('tpa_line_cd', ''))
 )
 return state_matches and date_matches and line_matches

def extract_X_coverage(coverage_str):
 """
 Extract standard coverage label. Handles distinctions for 'Excess' lines.
 Examples:
 '10 -Workers Compensation - X' → 'Workers Compensation'
 '30 -Auto Liability - X' → 'Auto Liability'
 """
 if pd.isna(coverage_str):
 return ""
 s = str(coverage_str).strip()
# Pattern: num - Coverage - X (or X) [optional 'Excess']
 match = re.match(r"^\d+\s*-\s*([^-]+)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if match:
 cov = match.group(1).strip()
 excess = match.group(2)
 if excess:
 cov += " Excess"
 return cov
# Handle other variant, e.g. missing numeric prefix, or 'X' capitalization
 s = re.sub(r"^\d+\s*-\s*", "", s)
 excess_match = re.search(r"(.*?)\s*-\s*X(?:\s*(Excess))?$", s, re.IGNORECASE)
 if excess_match:
 cov = excess_match.group(1).strip()
 excess = excess_match.group(2)
 if excess:
 cov += " Excess"
 return cov
# Fallback: just before any dash or trailing 'X' text
 s = re.sub(r"-\s*X.*$", "", s, flags=re.IGNORECASE).strip()
 return s
def get_base_claim(claim_no):
 """
 Extracts the BASE claim:
 1. If it ends with letters + digits (e.g., …GB01), strip trailing digits (result: …GB)
 2. Else if it ends with a single letter (e.g., …49A), strip trailing letter (result: …49)
 3. Finally, if there is a dash ('-'), take the portion before it
 """
 claim_core = str(claim_no).strip()
 # Step 1: Remove trailing digits if immediately following uppercase letters (e.g. GB01 -> GB)
 m = re.match(r'^(.*?[A-Z]+)[0–9]+$', claim_core, re.IGNORECASE)
 if m:
 claim_core = m.group(1)
 # Step 2: Remove single trailing letter (e.g. …49A -> …49)
 elif re.match(r'.*[A-Z]$', claim_core, re.IGNORECASE):
 claim_core = re.sub(r'([A-Z])$', '', claim_core, flags=re.IGNORECASE)
 # Step 3: If dash present, take portion before dash
 if '-' in claim_core:
 claim_core = claim_core.split('-')[0].strip()
 return claim_core
